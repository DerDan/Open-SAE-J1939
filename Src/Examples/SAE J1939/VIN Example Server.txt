/* This is the example of an ECU which is capable of responding to Vehicle Identification Number (VIN) requests (PGN: 0xFEEC).
 * This example has been built to use socketcan over a virtual CAN interface.
 * Pre-requisites:
 * - Have a virtual CAN interface working on your computer and pass its name to socketcan_setup below;
 * - Use option `-DOPENSAE_J1939_TARGET_PLATFORM=SOCKETCAN` to build with CMake;
 */

#include <stdio.h>

#include "Open_SAE_J1939/Open_SAE_J1939.h"
#include "Open_SAE_J1939/Structs.h"
#include "Hardware/SocketCAN.h"

/* This function reads the CAN traffic */
void Callback_Function_Traffic(J1939 *instance, uint32_t ID, uint8_t DLC, uint8_t data[], bool is_TX) {
	(void)instance;
    /* Print if it is TX or RX */
    printf("%s\t", is_TX ? "TX" : "RX");

    /* Print ID as hex */
    printf("%08X\t", ID);

    /* Print the data */
    uint8_t i;
    for (i = 0U; i < DLC; i++) {
        printf("%X\t", data[i]);
    }

    /* Print the non-data */
    for (i = DLC; i < 8U; i++) {
        printf("%X\t", 0U);
    }

    /* New line */
    printf("\n");
}

int main() {
    J1939 serverECU = { 0 };
    CAN_Context can_context;
    CAN_Context_Init(&can_context);
    serverECU.can_context = &can_context;
    if (socketcan_setup(&can_context, "can0") < 0) {
        return 1;
    }
    char vehicleVIN[MAX_VEHICLE_IDENTIFICATION] = "ABCDEFGHIJ0123456";
    serverECU.information_this_ECU.this_ECU_address = 0x6A;
    int i = 0;
    for (i = 0; i < MAX_VEHICLE_IDENTIFICATION; i++) {
        serverECU.information_this_ECU.this_vehicle_identification.vin[i] = (uint8_t)vehicleVIN[i];
    }

    Open_SAE_J1939_Closedown_ECU(&serverECU);
	
	/* Optional: it only prints to console the CAN traffic, it can be omitted if you use candump */
    CAN_Set_Callback_Functions(&can_context, NULL, NULL, Callback_Function_Traffic, NULL);

    Open_SAE_J1939_Startup_ECU(&serverECU);

    while (1) {
        Open_SAE_J1939_Listen_For_Messages(&serverECU);
    }

    socketcan_close(&can_context);
    return 0;
}
